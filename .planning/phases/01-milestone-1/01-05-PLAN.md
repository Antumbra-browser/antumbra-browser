  **welcome.js** — Screen navigation, pref writes, AntumbraMode wiring.

  ```js
  /* This Source Code Form is subject to the terms of the Mozilla Public
   * License, v. 2.0. If a copy of the MPL was not distributed with this
   * file, You can obtain one at http://mozilla.org/MPL/2.0/. */

  "use strict";

  // welcome.js runs in a privileged chrome page; Services is available.
  /* global Services */

  const VERTICAL_TABS_PREF = "sidebar.verticalTabs";
  const SIDEBAR_REVAMP_PREF = "sidebar.revamp";
  const TRR_URI_PREF = "network.trr.uri";
  const TRR_CUSTOM_PREF = "network.trr.custom_uri";
  const TRR_MODE_PREF = "network.trr.mode";
  const MODE_PREF = "antumbra.protection.mode";
  const FIRSTRUN_PREF = "antumbra.firstrun.completed";

  // Check if we've already completed first run (belt-and-suspenders).
  if (Services.prefs.getBoolPref(FIRSTRUN_PREF, false)) {
    // Already done; close this tab.
    window.close();
  }

  // Detect vertical tabs support.
  const verticalTabsAvailable = Services.prefs.getPrefType(VERTICAL_TABS_PREF) !== 0;
  if (!verticalTabsAvailable) {
    document.getElementById("tabs-unavailable").classList.remove("hidden");
    document.querySelectorAll('input[name="tabs"]').forEach(r => r.disabled = true);
  }

  // Load DNS resolvers from dns-resolvers.json.
  async function loadResolvers() {
    const resp = await fetch("chrome://antumbra/content/dns-resolvers.json");
    const data = await resp.json();
    const list = document.getElementById("resolver-list");
    data.resolvers.forEach(resolver => {
      const label = document.createElement("label");
      label.className = "resolver-option";
      label.htmlFor = `resolver-${resolver.id}`;
      const input = document.createElement("input");
      input.type = "radio";
      input.name = "resolver";
      input.id = `resolver-${resolver.id}`;
      input.value = resolver.id;
      if (resolver.default) input.checked = true;
      const div = document.createElement("div");
      div.className = "resolver-content";
      div.innerHTML = `
        <strong>${resolver.name}</strong>
        <span class="resolver-jurisdiction">${resolver.jurisdiction}</span>
        <p>${resolver.description}</p>
        <small>Logging: ${resolver.logging}</small>
      `;
      label.appendChild(input);
      label.appendChild(div);
      list.appendChild(label);
      // Show custom URL field when custom is selected.
      if (resolver.id === "custom") {
        input.addEventListener("change", () => {
          document.getElementById("custom-dns-field").classList.remove("hidden");
        });
      } else {
        input.addEventListener("change", () => {
          document.getElementById("custom-dns-field").classList.add("hidden");
        });
      }
    });
  }
  loadResolvers();

  // Screen navigation.
  document.querySelectorAll(".btn-next").forEach(btn => {
    btn.addEventListener("click", () => {
      const nextId = btn.dataset.next;
      document.querySelectorAll(".screen").forEach(s => s.classList.add("hidden"));
      document.getElementById(nextId).classList.remove("hidden");
    });
  });

  // Primary password screen.
  document.getElementById("btn-set-password")?.addEventListener("click", () => {
    // Open the primary password dialog via the chrome API.
    window.openDialog(
      "chrome://mozapps/content/preferences/changemp.xhtml",
      "changemp",
      "chrome,centerscreen,resizable=no"
    );
    // After the dialog, advance to the next screen.
    document.querySelectorAll(".screen").forEach(s => s.classList.add("hidden"));
    document.getElementById("screen-updates").classList.remove("hidden");
  });
  document.getElementById("btn-skip-password")?.addEventListener("click", () => {
    document.querySelectorAll(".screen").forEach(s => s.classList.add("hidden"));
    document.getElementById("screen-updates").classList.remove("hidden");
  });

  // Finish: write all choices to prefs.
  document.getElementById("btn-finish")?.addEventListener("click", () => {
    // 1. Protection mode.
    const mode = document.querySelector('input[name="mode"]:checked')?.value ?? "standard";
    const { AntumbraMode } = ChromeUtils.importESModule(
      "resource:///modules/AntumbraMode.sys.mjs"
    );
    AntumbraMode.applyMode(mode);

    // 2. DNS resolver.
    const resolverId = document.querySelector('input[name="resolver"]:checked')?.value ?? "quad9";
    if (resolverId === "system") {
      Services.prefs.setIntPref(TRR_MODE_PREF, 0); // System resolver: disable TRR.
    } else if (resolverId === "custom") {
      const customUrl = document.getElementById("custom-dns-url")?.value?.trim();
      if (customUrl) {
        Services.prefs.setStringPref(TRR_URI_PREF, customUrl);
        Services.prefs.setStringPref(TRR_CUSTOM_PREF, customUrl);
        Services.prefs.setIntPref(TRR_MODE_PREF, 2);
      }
    } else {
      // Named resolver: look up the URI from the loaded data.
      const resolverLabel = document.getElementById(`resolver-${resolverId}`);
      // URI was set as a data attribute during loadResolvers; retrieve it.
      // For simplicity: hardcode the URIs here matching dns-resolvers.json.
      const RESOLVER_URIS = {
        quad9:       "https://dns.quad9.net/dns-query",
        mullvad:     "https://dns.mullvad.net/dns-query",
        nextdns:     "https://dns.nextdns.io",
        cloudflare:  "https://cloudflare-dns.com/dns-query",
      };
      const uri = RESOLVER_URIS[resolverId];
      if (uri) {
        Services.prefs.setStringPref(TRR_URI_PREF, uri);
        Services.prefs.setStringPref(TRR_CUSTOM_PREF, uri);
        Services.prefs.setIntPref(TRR_MODE_PREF, 2);
      }
    }

    // 3. Tab orientation (if the pref exists in ESR 153).
    if (verticalTabsAvailable) {
      const tabChoice = document.querySelector('input[name="tabs"]:checked')?.value ?? "vertical";
      const useVertical = tabChoice === "vertical";
      Services.prefs.setBoolPref(VERTICAL_TABS_PREF, useVertical);
      Services.prefs.setBoolPref(SIDEBAR_REVAMP_PREF, useVertical);
    }

    // 4. Mark first run complete.
    Services.prefs.setBoolPref(FIRSTRUN_PREF, true);

    // Navigate to the new tab page.
    window.location.href = "about:newtab";
  });
  ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| welcome.js → Services.prefs | The wizard writes directly to the pref store from a privileged chrome page; incorrect writes could weaken the privacy baseline. |
| dns-resolvers.json → user's DNS choice | The resolver list is an in-tree static file; a tampered list would direct users to a malicious resolver. |
| chrome:// page → AntumbraMode.sys.mjs | The wizard calls AntumbraMode.applyMode() with the user-selected mode; a wrong value silently degrades protections. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-05-01 | Tampering | dns-resolvers.json URIs | mitigate | dns-resolvers.json is committed to the antumbra-browser repo; any change to resolver URIs is visible in git history and CI diff review. Resolver URIs are also hardcoded in welcome.js as a cross-check. |
| T-01-05-02 | Information Disclosure | DNS resolver choice sent to resolver | accept | The user explicitly chose the resolver in the wizard. The jurisdiction and logging policy are disclosed inline per ARCHITECTURE.md 5.7. |
| T-01-05-03 | Spoofing | welcome.html external links open target="_blank" | mitigate | Links open to antumbrabrowser.org (our own domain) and github.com/antumbra-browser. rel="noopener" is added in the final HTML. |
| T-01-05-04 | Tampering | firstrun.completed pref can be reset | accept | A user resetting firstrun.completed to false re-sees the wizard, which is a self-service action and not a security issue. The wizard only sets prefs; it does not install software or request elevated permissions. |
| T-01-05-05 | Information Disclosure | mode choice visible via antumbra-mode attribute on :root | accept | The mode is user-chosen and displayed in the UI. Web content can read it via getComputedStyle or attribute inspection. Accepted per T-01-04-04 rationale. |

Note: No user data is uploaded in Milestone 1. The wizard runs locally; the only network effect is writing the DNS resolver pref, which takes effect on subsequent DNS lookups.
</threat_model>
